function confirmDelete() {

title: "¿Estas seguro?",

confirmButtonText: "Si, eliminar!",
customClass: {
title: 'text-black',
confirmButton: 'btn-danger',
cancelButton: 'btn-primary',
}
}).then((result) => {
if (result.isConfirmed) {
document.getElementById('delete-form').submit();
}
});
}
</script>

Esta es la parte del controllador que se encarga de eliminarlo

public function destroy(RequisitionType $requisitionType)
{

$requisitionType->delete();

session()->flash('swal', [
'title' => '!Bien hecho!',
'text' => 'Tipo de requisición eliminado correctamente',
'icon' => 'success',
'customClass' => [
'title' => 'text-black',
'confirmButton' => 'btn-primary',
]
]);

return redirect()->route('requisition-types.index');
}
